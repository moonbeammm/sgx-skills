
1、如果修复的修改对一个方法改动量较大，则对整个方法做开关
```
// 原方法
func legacy_append(from: Int, to: Int) {

}

// 增加开关方式
func append(from: Int, to: Int) {
	if FF.StoryListAppendConsistencyFix {
		fixed_append(from: from, to: to)
	} else {
		legacy_append(from: from, to: to)
	}
}

private func fixed_append(from: Int, to: Int) {

}

private func legacy_append(from: Int, to: Int) {

}
```
